# Laufzeitabhaengigkeiten des Backends.
#
# Bewusst NICHT in pyproject.toml und pdm.lock: Der Upstream aktualisiert seine
# Abhaengigkeiten automatisiert (.github/workflows/update-python-deps.yml) und
# erzeugt pdm.lock dabei neu. Stuenden unsere Pakete dort, gaebe es bei jedem
# dieser Laeufe einen Merge-Konflikt in einer generierten Datei - der teuerste
# Konflikttyp ueberhaupt. Die Trennung kostet eine eigene Datei und spart
# dauerhaft Reibung.
#
# Lizenzen dieser Pakete gehoeren nach docs/LIZENZEN.md, sobald sie ueber das
# Geruest hinaus benutzt werden.

cryptography>=44.0,<50.0
fastapi>=0.115,<1.0
# HTTP-Client fuer das KI-Modul (AP-4.1). Muss asynchron sein: Ein Aufruf beim
# Anbieter dauert Sekunden, und ein blockierender Aufruf haelt in einer
# asyncio-Anwendung nicht nur diese Anfrage auf, sondern alle. urllib aus der
# Standardbibliothek - wie in katalog/daten.py benutzt - kann das nicht.
# httpx liegt ohnehin in jeder Testumgebung, weil FastAPIs TestClient darauf
# aufbaut; neu ist damit nur die Zeile, nicht die Bibliothek.
httpx>=0.27,<1.0
# Bilder verkleinern und von Metadaten befreien, bevor sie den Rechner
# verlassen (AP-4.7). Ohne eine Bildbibliothek liesse sich weder das eine noch
# das andere machen - und EXIF-Daten eines Handyfotos enthalten regelmaessig
# die GPS-Koordinaten des Aufnahmeorts, also die Wohnadresse.
pillow>=11.0,<13.0
# Aufraeumen verwaister Browserprozesse (AP-1.9). Der Bot bringt psutil
# ohnehin mit; hier steht es, weil das Backend es selbst importiert und die
# Abhaengigkeit nicht stillschweigend vom Bot geborgt werden soll.
psutil>=6.0
# Fuer Datei-Uploads (AP-2.6). FastAPI braucht es, um multipart/form-data zu
# lesen; ohne die Zeile scheitert der Endpunkt beim Start mit einer
# Fehlermeldung, die nach einem Programmfehler aussieht.
# Lizenz am 2026-08-25 ueber die PyPI-API geprueft: Apache-2.0, vertraeglich.
python-multipart>=0.0.20,<1.0
# Erzeugt die config.yaml des Bots. Gleiche Bibliothek wie der Bot selbst,
# damit Formatierung und Sonderfaelle uebereinstimmen.
ruamel.yaml>=0.18,<1.0
uvicorn[standard]>=0.32,<1.0
